1//! Shell execution for agents (Phase 1–2 of AGENT_SHELL_EVOLUTION_PLAN).
2//! sh::run(cmd) runs a command in a subprocess; trust config (off/sandboxed/confirmed/trusted)
3//! is read from env DAL_AGENT_SHELL_TRUST, agent.toml \[agent.sh\], or dal.toml \[agent.sh\].
4//! When owner_principal_id is set (env DAL_AGENT_OWNER_PRINCIPAL or \[agent\] owner_principal_id),
5//! key::check("sh", "run", principal_id) is consulted first; if key allows, run proceeds; if key
6//! denies or no key exists, we fall back to \[agent.sh\] config.
7

14/// Trust level for sh execution. Precedence: env DAL_AGENT_SHELL_TRUST → agent.toml / dal.toml \[agent.sh\] → default "sandboxed".
15#[derive(Debug, Clone, PartialEq, Eq)]
16pub enum TrustLevel {
17    Off,
18    Sandboxed,
19    Confirmed,
20    Trusted,
21}
22
23impl TrustLevel {
24    fn from_str(s: &str) -> Self {
25        match s.trim().to_lowercase().as_str() {
26            "off" => TrustLevel::Off,
27            "confirmed" => TrustLevel::Confirmed,
28            "trusted" => TrustLevel::Trusted,
29            _ => TrustLevel::Sandboxed,
30        }
31    }
32}
33
34impl std::fmt::Display for TrustLevel {
35    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
36        match self {
37            TrustLevel::Off => write!(f, "off"),
38            TrustLevel::Sandboxed => write!(f, "sandboxed"),
39            TrustLevel::Confirmed => write!(f, "confirmed"),
40            TrustLevel::Trusted => write!(f, "trusted"),
41        }
42    }
43}

45/// Config for [agent.sh]. Loaded from agent.toml or dal.toml or env.
46#[derive(Debug, Clone)]
47pub struct ShConfig {
48    pub trust: TrustLevel,
49    pub forbidden_patterns: Vec<String>,
50    pub allowed_prefixes: Option<Vec<String>>,
51}
52
53impl Default for ShConfig {
54    fn default() -> Self {
55        Self {
56            trust: TrustLevel::Sandboxed,
57            forbidden_patterns: Vec::new(),
58            allowed_prefixes: None,
59        }
60    }
61}

63/// Load [agent.sh] from a TOML table (e.g. from agent.toml root or dal.toml under ["agent"]["sh"]).
64fn parse_agent_sh_from_table(table: &toml::Table) -> Option<ShConfig> {
65    let agent_sh = table
66        .get("agent")
67        .and_then(|v| v.as_table())
68        .and_then(|t| t.get("sh"))
69        .and_then(|v| v.as_table());
70    let table = agent_sh.or_else(|| table.get("sh").and_then(|v| v.as_table()))?;
71    let trust = table
72        .get("trust")
73        .and_then(|v| v.as_str())
74        .map(TrustLevel::from_str)
75        .unwrap_or(TrustLevel::Sandboxed);
76    let forbidden_patterns: Vec<String> = table
77        .get("forbidden_patterns")
78        .and_then(|v| v.as_array())
79        .map(|a| {
80            a.iter()
81                .filter_map(|v| v.as_str().map(String::from))
82                .collect()
83        })
84        .unwrap_or_default();
85    let allowed_prefixes = table
86        .get("allowed_prefixes")
87        .and_then(|v| v.as_array())
88        .map(|a| {
89            a.iter()
90                .filter_map(|v| v.as_str().map(String::from))
91                .collect()
92        });
93    Some(ShConfig {
94        trust,
95        forbidden_patterns,
96        allowed_prefixes,
97    })
98}

100/// Load sh config: dal.toml then agent.toml (agent.toml overrides); env DAL_AGENT_SHELL_TRUST overrides trust.
101pub fn load_sh_config() -> ShConfig {
102    let cwd = env::current_dir().unwrap_or_else(|_| std::path::PathBuf::from("."));
103    let mut config = ShConfig::default();
104
105    if let Ok(content) = std::fs::read_to_string(cwd.join("dal.toml")) {
106        if let Ok(table) = content.parse::<toml::Table>() {
107            if let Some(c) = parse_agent_sh_from_table(&table) {
108                config = c;
109            }
110        }
111    }
112    if let Ok(content) = std::fs::read_to_string(cwd.join("agent.toml")) {
113        if let Ok(table) = content.parse::<toml::Table>() {
114            if let Some(c) = parse_agent_sh_from_table(&table) {
115                config = c;
116            }
117        }
118    }
119
120    if let Ok(trust_env) = env::var("DAL_AGENT_SHELL_TRUST") {
121        config.trust = TrustLevel::from_str(&trust_env);
122    }
123    config
124}

126/// Human-readable constraints description for agent prompt context (P4). Shell trust and forbidden patterns from [agent.sh].
127pub fn constraints_description_for_prompt(config: &ShConfig) -> String {
128    let mut lines = vec![format!("Shell: {}.", config.trust)];
129    if !config.forbidden_patterns.is_empty() {
130        lines.push(format!(
131            "Forbidden command patterns: {}.",
132            config.forbidden_patterns.join(", ")
133        ));
134    }
135    lines.join(" ")
136}

138/// Load owner principal ID for key::check. Precedence: DAL_AGENT_OWNER_PRINCIPAL → agent.toml \[agent\] owner_principal_id → dal.toml \[agent\] owner_principal_id.
139pub fn load_owner_principal_id() -> Option<String> {
140    if let Ok(id) = env::var("DAL_AGENT_OWNER_PRINCIPAL") {
141        if !id.trim().is_empty() {
142            return Some(id.trim().to_string());
143        }
144    }
145    let cwd = env::current_dir().unwrap_or_else(|_| std::path::PathBuf::from("."));
146    for file in ["agent.toml", "dal.toml"] {
147        let content = std::fs::read_to_string(cwd.join(file)).ok()?;
148        let table = content.parse::<toml::Table>().ok()?;
149        let agent = table.get("agent").and_then(|v| v.as_table())?;
150        let id = agent.get("owner_principal_id").and_then(|v| v.as_str())?;
151        if !id.trim().is_empty() {
152            return Some(id.trim().to_string());
153        }
154    }
155    None
156}

158/// Check command against config: returns Err if not allowed.
159fn check_allowed(cmd: &str, config: &ShConfig) -> Result<(), String> {
160    if config.trust == TrustLevel::Off {
161        return Err("sh is disabled by config (trust=off). Set DAL_AGENT_SHELL_TRUST or [agent.sh] trust to allow.".to_string());
162    }
163    if config.trust == TrustLevel::Confirmed {
164        if env::var("DAL_AGENT_SHELL_CONFIRM").ok().as_deref() != Some("1") {
165            return Err(
166                "sh::run: confirmed mode requires approval. Set DAL_AGENT_SHELL_CONFIRM=1 to run without prompt, or use trust=trusted/sandboxed in config.".to_string()
167            );
168        }
169    }
170    let cmd_trim = cmd.trim();
171    for pattern in &config.forbidden_patterns {
172        if cmd_trim.contains(pattern) {
173            return Err(format!(
174                "command forbidden by config (forbidden_patterns): {}",
175                pattern
176            ));
177        }
178    }
179    if let Some(ref prefixes) = config.allowed_prefixes {
180        if !prefixes.is_empty() {
181            let first_word = cmd_trim.split_whitespace().next().unwrap_or("");
182            let allowed = prefixes
183                .iter()
184                .any(|p| first_word.starts_with(p) || cmd_trim.starts_with(p));
185            if !allowed {
186                return Err(format!(
187                    "command not in allowed_prefixes (config): {:?}",
188                    prefixes
189                ));
190            }
191        }
192    }
193    Ok(())
194}

196/// Run a shell command. Respects [agent.sh], DAL_AGENT_SHELL_TRUST, and key::check when owner_principal_id is set. Returns a map: stdout, stderr, exit_code.
197pub fn run(cmd: &str) -> Result<Value, String> {
198    let config = load_sh_config();
199    if let Some(principal_id) = load_owner_principal_id() {
200        let req = CapabilityRequest {
201            resource: "sh".to_string(),
202            operation: "run".to_string(),
203            principal_id: principal_id.clone(),
204        };
205        if key::check(req).unwrap_or(false) {
206            // Key allows: still apply config (forbidden_patterns, etc.)
207        } else {
208            // Key denies or no key: fall back to [agent.sh] config (check_allowed below)
209        }
210    }
211    check_allowed(cmd, &config)?;
212    let (stdout, stderr, exit_code) = run_impl(cmd)?;
213    let mut map = HashMap::new();
214    map.insert("stdout".to_string(), Value::String(stdout));
215    map.insert("stderr".to_string(), Value::String(stderr));
216    map.insert("exit_code".to_string(), Value::Int(exit_code));
217    Ok(Value::Map(map))
218}
219
220fn run_impl(cmd: &str) -> Result<(String, String, i64), String> {
221    let output = if cfg!(unix) {
222        Command::new("sh")
223            .arg("-c")
224            .arg(cmd)
225            .output()
226            .map_err(|e| e.to_string())?
227    } else if cfg!(windows) {
228        Command::new("cmd")
229            .args(["/C", cmd])
230            .output()
231            .map_err(|e| e.to_string())?
232    } else {
233        return Err("Unsupported platform for sh::run".to_string());
234    };
235
236    let stdout = String::from_utf8_lossy(&output.stdout).to_string();
237    let stderr = String::from_utf8_lossy(&output.stderr).to_string();
238    let exit_code = output.status.code().unwrap_or(-1);
239    Ok((stdout, stderr, exit_code as i64))
240}

253    struct EnvVarGuard {
254        key: &'static str,
255        old: Option<String>,
256    }
257
258    impl EnvVarGuard {
259        fn set(key: &'static str, value: Option<&str>) -> Self {
260            let old = std::env::var(key).ok();
261            if let Some(v) = value {
262                std::env::set_var(key, v);
263            } else {
264                std::env::remove_var(key);
265            }
266            Self { key, old }
267        }
268    }
269
270    impl Drop for EnvVarGuard {
271        fn drop(&mut self) {
272            if let Some(v) = &self.old {
273                std::env::set_var(self.key, v);
274            } else {
275                std::env::remove_var(self.key);
276            }
277        }
278    }
279
280    struct CwdGuard {
281        old: std::path::PathBuf,
282    }
283
284    impl CwdGuard {
285        fn set_to(path: &std::path::Path) -> Self {
286            let old = std::env::current_dir().expect("cwd");
287            std::env::set_current_dir(path).expect("set cwd");
288            Self { old }
289        }
290    }
291
292    impl Drop for CwdGuard {
293        fn drop(&mut self) {
294            std::env::set_current_dir(&self.old).expect("restore cwd");
295        }
296    }
