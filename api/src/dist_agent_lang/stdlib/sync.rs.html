5fn value_to_serde_json(v: &Value) -> serde_json::Value {
6    match v {
7        Value::Int(n) => serde_json::json!(n),
8        Value::Float(f) => serde_json::json!(f),
9        Value::String(s) => serde_json::json!(s),
10        Value::Bool(b) => serde_json::json!(b),
11        Value::Null => serde_json::Value::Null,
12        Value::List(arr) => serde_json::Value::Array(arr.iter().map(value_to_serde_json).collect()),
13        Value::Map(m) => serde_json::Value::Object(
14            m.iter()
15                .map(|(k, v)| (k.clone(), value_to_serde_json(v)))
16                .collect(),
17        ),
18        Value::Struct(_, m) => serde_json::Value::Object(
19            m.iter()
20                .map(|(k, v)| (k.clone(), value_to_serde_json(v)))
21                .collect(),
22        ),
23        Value::Array(arr) => {
24            serde_json::Value::Array(arr.iter().map(value_to_serde_json).collect())
25        }
26        _ => serde_json::Value::String(v.to_string()),
27    }
28}

30#[cfg(feature = "http-interface")]
31fn serde_json_to_value(j: &serde_json::Value) -> Value {
32    match j {
33        serde_json::Value::Null => Value::Null,
34        serde_json::Value::Bool(b) => Value::Bool(*b),
35        serde_json::Value::Number(n) => {
36            if let Some(i) = n.as_i64() {
37                Value::Int(i)
38            } else if let Some(f) = n.as_f64() {
39                Value::Float(f)
40            } else {
41                Value::String(n.to_string())
42            }
43        }
44        serde_json::Value::String(s) => Value::String(s.clone()),
45        serde_json::Value::Array(arr) => Value::List(arr.iter().map(serde_json_to_value).collect()),
46        serde_json::Value::Object(obj) => {
47            let m: HashMap<String, Value> = obj
48                .iter()
49                .map(|(k, v)| (k.clone(), serde_json_to_value(v)))
50                .collect();
51            Value::Map(m)
52        }
53    }
54}

56/// Sync ABI - Interface for data synchronization
57///
58/// This provides a namespace-based approach to sync operations:
59/// - sync::push(data, target) - Push data to target location
60/// - sync::pull(source, filters) - Pull data from source
61
62#[derive(Debug, Clone)]
63pub struct SyncTarget {
64    pub location: String,
65    pub protocol: String,
66    pub credentials: Option<HashMap<String, String>>,
67    pub compression: bool,
68}

70#[derive(Debug, Clone)]
71pub struct SyncFilters {
72    pub data_type: Option<String>,
73    pub date_range: Option<(i64, i64)>,
74    pub tags: Vec<String>,
75    pub max_size: Option<i64>,
76}

78impl SyncTarget {
79    pub fn new(location: String, protocol: String) -> Self {
80        Self {
81            location,
82            protocol,
83            credentials: None,
84            compression: false,
85        }
86    }
87
88    pub fn with_credentials(mut self, credentials: HashMap<String, String>) -> Self {
89        self.credentials = Some(credentials);
90        self
91    }
92
93    pub fn with_compression(mut self, compression: bool) -> Self {
94        self.compression = compression;
95        self
96    }
97}

106        }
107    }

136/// Push data to target location. When http-interface and protocol is http/https, POSTs data as JSON.
137pub fn push(data: HashMap<String, Value>, target: SyncTarget) -> Result<bool, String> {
138    #[cfg(feature = "http-interface")]
139    if target.protocol == "http" || target.protocol == "https" {
140        let body = serde_json::Value::Object(
141            data.iter()
142                .map(|(k, v)| (k.clone(), value_to_serde_json(v)))
143                .collect(),
144        );
145        let client = reqwest::blocking::Client::builder()
146            .timeout(std::time::Duration::from_secs(30))
147            .build()
148            .map_err(|e| e.to_string())?;
149        let resp = client
150            .post(&target.location)
151            .json(&body)
152            .send()
153            .map_err(|e| e.to_string())?;
154        if resp.status().is_success() {
155            return Ok(true);
156        }
157        return Err(format!("HTTP push failed: {}", resp.status()));
158    }
159
160    // Fallback: mock by protocol
161    match target.protocol.as_str() {
162        "http" | "https" => {
163            if target.location.contains("api.example.com") {
164                Ok(true)
165            } else {
166                Err("HTTP push failed: Invalid endpoint".to_string())
167            }
168        }
169        "ftp" => {
170            if target.location.contains("ftp.example.com") {
171                Ok(true)
172            } else {
173                Err("FTP push failed: Invalid endpoint".to_string())
174            }
175        }
176        "s3" => {
177            if target.location.contains("s3.amazonaws.com") {
178                Ok(true)
179            } else {
180                Err("S3 push failed: Invalid endpoint".to_string())
181            }
182        }
183        _ => Err(format!("Unsupported protocol: {}", target.protocol)),
184    }
185}

187/// Pull data from source. When http-interface and source is a URL (http/https), GETs and parses JSON to map.
188pub fn pull(source: &str, filters: SyncFilters) -> Result<(HashMap<String, Value>, bool), String> {
189    #[cfg(feature = "http-interface")]
190    if source.starts_with("http://") || source.starts_with("https://") {
191        let client = reqwest::blocking::Client::builder()
192            .timeout(std::time::Duration::from_secs(30))
193            .build()
194            .map_err(|e| e.to_string())?;
195        let resp = client.get(source).send().map_err(|e| e.to_string())?;
196        if !resp.status().is_success() {
197            return Err(format!("HTTP pull failed: {}", resp.status()));
198        }
199        let json: serde_json::Value = resp.json().map_err(|e| e.to_string())?;
200        let value = serde_json_to_value(&json);
201        if let Value::Map(m) = value {
202            return Ok((m, true));
203        }
204        // Single value: wrap in a map
205        let mut data = HashMap::new();
206        data.insert("data".to_string(), value);
207        return Ok((data, true));
208    }
209
210    // Fallback: mock sources
211    match source {
212        "database" => {
213            let mut data = HashMap::new();
214            data.insert(
215                "user_123".to_string(),
216                Value::String("John Doe".to_string()),
217            );
218            data.insert(
219                "user_456".to_string(),
220                Value::String("Jane Smith".to_string()),
221            );
222
223            if let Some(data_type) = &filters.data_type {
224                if data_type == "users" {
225                    Ok((data, true))
226                } else {
227                    Err("Data type filter not supported".to_string())
228                }
229            } else {
230                Ok((data, true))
231            }
232        }
233        "api" => {
234            let mut data = HashMap::new();
235            data.insert("price_btc".to_string(), Value::Int(45000));
236            data.insert("price_eth".to_string(), Value::Int(3200));
237
238            Ok((data, true))
239        }
240        _ => Err(format!("Unknown source: {}", source)),
241    }
242}
